        (function() {
            var spriteUrl = 'https://example.com/assets/svg/icons.svg?v=1769662442';
            var inserted = false;


            function insertSprite(svg) {
                if (!svg || inserted) return;
                var div = document.createElement('div');

                div.style.position = 'absolute';
                div.style.width = '0';
                div.style.height = '0';
                div.style.overflow = 'hidden';
                div.setAttribute('aria-hidden', 'true');

                try {
                    var parser = new DOMParser();
                    var doc = parser.parseFromString(svg, 'image/svg+xml');
                    var svgEl = doc.documentElement;
                    if (svgEl && svgEl.nodeName.toLowerCase() === 'svg') {
                        div.appendChild(document.importNode(svgEl, true));
                    } else {
                        div.innerHTML = svg;
                    }
                } catch (e) {
                    div.innerHTML = svg;
                }
                div.setAttribute('id', 'svg-sprite-container');
                var target = document.body || document.documentElement;
                target.insertBefore(div, target.firstChild);
                inserted = true;
            }

            function loadSprite() {
                fetch(spriteUrl, {
                    method: 'GET',
                    cache: 'default',
                    credentials: 'same-origin'
                })
                    .then(function(response) {
                        if (!response.ok) {
                            console.warn('SVG sprite failed to load:', response.status);
                            return null;
                        }
                        return response.text();
                    })
                    .then(function(svg) {
                        if (svg) {
                            insertSprite(svg);
                        }
                    })
                    .catch(function(error) {
                        if (window.location.hostname === 'localhost' || window.location.hostname === '127.0.0.1') {
                            console.warn('SVG sprite loading error:', error);
                        }
                    });
            }
            if (document.readyState === 'loading') {
                document.addEventListener('DOMContentLoaded', loadSprite);
            } else {
                loadSprite();
            }
        })();

        (function(){
            const modal = document.getElementById('howToPlayModal');
            if(!modal) return;

            const panes   = Array.from(modal.querySelectorAll('.howto-pane'));
            const steps   = Array.from(modal.querySelectorAll('#howto-stepper .step'));
            const bar     = modal.querySelector('#howto-progressbar');
            const btnPrev = modal.querySelector('#howto-prev');
            const btnNext = modal.querySelector('#howto-next');
            const btnSkip = modal.querySelector('#howto-skip');
            const btnDone = modal.querySelector('#howto-done');

            const bsModal = () => bootstrap.Modal.getInstance(modal) || new bootstrap.Modal(modal);
            const total = panes.length;
            let idx = 0; // 0..total-1

            function render(){
                panes.forEach((p,i)=>p.classList.toggle('d-none', i!==idx));
                steps.forEach((s,i)=>{
                    s.classList.toggle('active', i===idx);
                    s.classList.toggle('completed', i<idx);
                });
                // progress: completed-only; give step 1 a tiny head so it doesn't look empty
                let pct = (total > 1) ? (idx / (total - 1)) * 100 : 100;
                if (idx === 0) {
                    bar.style.width = '6px';            // tiny head
                } else {
                    bar.style.width = pct + '%';
                }
                bar.setAttribute('aria-valuenow', Math.round(pct));



                // Back behaves as Close on first step
                btnPrev.textContent = (idx===0) ? 'Schließen' : 'Zurück';
                btnPrev.dataset.close = (idx===0) ? '1' : '';
                // Enable/disable back button - should always be enabled
                btnPrev.disabled = false;

                btnNext.classList.toggle('d-none', idx===total-1);
                btnDone.classList.toggle('d-none', idx!==total-1);
                btnSkip.classList.toggle('d-none', idx>0);
            }

            function next(){ if(idx<total-1){ idx++; render(); } }
            function prevOrClose(){
                if(idx===0){ bsModal().hide(); return; }
                idx--; render();
            }

            btnNext.addEventListener('click', next);
            btnPrev.addEventListener('click', prevOrClose);
            btnSkip.addEventListener('click', ()=>{ idx = total-1; render(); });

            // step navigation
            steps.forEach((step, i) => {
                step.addEventListener('click', () => {
                    idx = i;
                    render();
                });
            });

            // keyboard navigation
            modal.addEventListener('keydown', (e)=>{
                if(e.key==='ArrowRight') next();
                if(e.key==='ArrowLeft')  prevOrClose();
            });

            // gentle entrance + reset
            modal.addEventListener('shown.bs.modal', ()=>{ idx = 0; render(); });
            modal.addEventListener('hidden.bs.modal', ()=>{ idx = 0; render(); });
        })();
    </script>
