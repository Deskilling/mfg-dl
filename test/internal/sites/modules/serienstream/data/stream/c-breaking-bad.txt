        (function() {
            var spriteUrl = 'https://example.com/assets/svg/icons.svg?v=1769662442';
            var inserted = false;


            function insertSprite(svg) {
                if (!svg || inserted) return;
                var div = document.createElement('div');

                div.style.position = 'absolute';
                div.style.width = '0';
                div.style.height = '0';
                div.style.overflow = 'hidden';
                div.setAttribute('aria-hidden', 'true');

                try {
                    var parser = new DOMParser();
                    var doc = parser.parseFromString(svg, 'image/svg+xml');
                    var svgEl = doc.documentElement;
                    if (svgEl && svgEl.nodeName.toLowerCase() === 'svg') {
                        div.appendChild(document.importNode(svgEl, true));
                    } else {
                        div.innerHTML = svg;
                    }
                } catch (e) {
                    div.innerHTML = svg;
                }
                div.setAttribute('id', 'svg-sprite-container');
                var target = document.body || document.documentElement;
                target.insertBefore(div, target.firstChild);
                inserted = true;
            }

            function loadSprite() {
                fetch(spriteUrl, {
                    method: 'GET',
                    cache: 'default',
                    credentials: 'same-origin'
                })
                    .then(function(response) {
                        if (!response.ok) {
                            console.warn('SVG sprite failed to load:', response.status);
                            return null;
                        }
                        return response.text();
                    })
                    .then(function(svg) {
                        if (svg) {
                            insertSprite(svg);
                        }
                    })
                    .catch(function(error) {
                        if (window.location.hostname === 'localhost' || window.location.hostname === '127.0.0.1') {
                            console.warn('SVG sprite loading error:', error);
                        }
                    });
            }
            if (document.readyState === 'loading') {
                document.addEventListener('DOMContentLoaded', loadSprite);
            } else {
                loadSprite();
            }
        })();

        <script>
            (function() {
                function initDescriptionToggle() {
                    const descId = 'series-desc-420';
                    const wrapper = document.getElementById(descId);
                    if (!wrapper) {
                        setTimeout(initDescriptionToggle, 100);
                        return;
                    }

                    const textEl = wrapper.querySelector('.description-text');
                    const btn = wrapper.querySelector('.show-more-toggle');
                    const p = wrapper.querySelector('p');
                    if (!textEl || !p) return;
                    if (!btn) return;

                    const originalText = textEl.textContent.trim();
                    if (!originalText) return;

                    let isExpanded = false;

                    function checkAndTruncate() {
                        if (p.offsetWidth === 0) {
                            setTimeout(checkAndTruncate, 100);
                            return;
                        }

                        // First check if text needs truncation by comparing heights
                        const tempFull = document.createElement('div');
                        const computedStyle = window.getComputedStyle(p);
                        tempFull.style.cssText = 'position:absolute;visibility:hidden;width:' + p.offsetWidth + 'px;font-size:' + computedStyle.fontSize + ';font-family:' + computedStyle.fontFamily + ';line-height:' + computedStyle.lineHeight + ';padding:' + computedStyle.padding + ';white-space:normal;word-wrap:break-word;box-sizing:border-box;';
                        tempFull.textContent = originalText;
                        document.body.appendChild(tempFull);

                        const lineHeight = parseFloat(computedStyle.lineHeight) || 24;
                        const maxHeight = lineHeight * 3;
                        const needsTruncation = tempFull.offsetHeight > maxHeight;
                        document.body.removeChild(tempFull);

                        if (!needsTruncation) {
                            btn.style.display = 'none';
                            return;
                        }

                        // Text needs truncation, find the right point
                        const temp = document.createElement('div');
                        temp.style.cssText = tempFull.style.cssText;
                        document.body.appendChild(temp);

                        const btnText = 'Mehr anzeigen';
                        let left = 0;
                        let right = originalText.length;
                        let bestFit = '';

                        while (left <= right) {
                            const mid = Math.floor((left + right) / 2);
                            const testText = originalText.substring(0, mid) + '... ' + btnText;
                            temp.textContent = testText;

                            if (temp.offsetHeight <= maxHeight) {
                                bestFit = originalText.substring(0, mid);
                                left = mid + 1;
                            } else {
                                right = mid - 1;
                            }
                        }

                        document.body.removeChild(temp);

                        if (bestFit.length > 0 && bestFit.length < originalText.length) {
                            textEl.textContent = bestFit;
                            btn.style.display = 'inline';
                        } else {
                            btn.style.display = 'none';
                        }
                    }

                    function toggleDescription(e) {
                        e.preventDefault();
                        isExpanded = !isExpanded;

                        if (isExpanded) {
                            textEl.textContent = originalText;
                            btn.textContent = 'Weniger anzeigen';
                            p.classList.remove('clamped');
                        } else {
                            checkAndTruncate();
                            btn.textContent = 'Mehr anzeigen';
                            p.classList.add('clamped');
                        }
                    }

                    btn.addEventListener('click', toggleDescription);
                    setTimeout(checkAndTruncate, 150);

                    let resizeTimer;
                    window.addEventListener('resize', function() {
                        clearTimeout(resizeTimer);
                        resizeTimer = setTimeout(function() {
                            if (!isExpanded) {
                                checkAndTruncate();
                            }
                        }, 250);
                    });
                }

                if (document.readyState === 'loading') {
                    document.addEventListener('DOMContentLoaded', initDescriptionToggle);
                } else {
                    setTimeout(initDescriptionToggle, 50);
                }
            })();
        </script>

    <script>
        (function () {
            const modalEl = document.getElementById('trailerModal');
            if (!modalEl) return;

            const toEmbedUrl = (raw) => {
                try {
                    const u = new URL(raw);
                    const host = u.hostname.replace(/^www\./i, '');

                    // YouTube
                    if (host.includes('youtube.com') || host === 'youtu.be') {
                        // id from ?v=, /shorts/{id}, /embed/{id}, or youtu.be/{id}
                        let id = '';
                        if (host === 'youtu.be') {
                            id = u.pathname.replace('/', '').split('?')[0];
                        } else if (u.searchParams.get('v')) {
                            id = u.searchParams.get('v');
                        } else if (u.pathname.startsWith('/shorts/')) {
                            id = u.pathname.split('/')[2] || '';
                        } else if (u.pathname.startsWith('/embed/')) {
                            id = u.pathname.split('/')[2] || '';
                        } else if (u.pathname.startsWith('/live/')) {
                            id = u.pathname.split('/')[2] || '';
                        }
                        if (id) return `https://example.com/embed/${encodeURIComponent(id)}?autoplay=1&rel=0`;
                    }

                    // Vimeo
                    if (host.includes('vimeo.com')) {
                        // vimeo.com/{id} or vimeo.com/channels/.../{id}
                        const parts = u.pathname.split('/').filter(Boolean);
                        const id = parts.reverse().find(p => /^\d+$/.test(p));
                        if (id) return `https://example.com/video/${id}?autoplay=1`;
                    }

                    // Dailymotion
                    if (host.includes('dailymotion.com')) {
                        // dailymotion.com/video/{id}
                        const parts = u.pathname.split('/').filter(Boolean);
                        const idx = parts.indexOf('video');
                        if (idx !== -1 && parts[idx + 1]) {
                            const id = parts[idx + 1];
                            return `https://example.com/embed/video/${encodeURIComponent(id)}?autoplay=1`;
                        }
                    }

                    // If already an embeddable URL, allow it as-is with autoplay
                    if (/\/embed\//i.test(u.pathname)) {
                        if (!u.searchParams.has('autoplay')) {
                            u.searchParams.set('autoplay', '1');
                        }
                        return u.toString();
                    }

                    // Fallback: open raw URL in an iframe (may be blocked by X-Frame-Options)
                    return raw;
                } catch {
                    return raw;
                }
            };

            // When opening: build embed URL from the triggering button's data-trailer-url
            modalEl.addEventListener('show.bs.modal', (ev) => {
                const btn = ev.relatedTarget;
                const rawUrl = btn?.getAttribute('data-trailer-url');
                const iframe = document.getElementById('trailer-iframe');
                const openLink = document.getElementById('trailer-open-in-new-tab');
                if (!rawUrl || !iframe) return;

                iframe.src = toEmbedUrl(rawUrl);
                if (openLink) openLink.href = rawUrl;
            });

            // When closing: stop playback
            modalEl.addEventListener('hidden.bs.modal', () => {
                const iframe = document.getElementById('trailer-iframe');
                if (iframe) iframe.src = '';
            });
        })();
    </script>
